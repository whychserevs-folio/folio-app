import React from "react";
import {
  View,
  Text,
  Image,
  Pressable,
  StyleSheet,
} from "react-native";

const COLOURS = {
  bg: "#101012",
  shelf: "#1A1B1F",
  shelfEdge: "#2E3238",
  text: "#F0EBE7",
  muted: "#A39B9B",
  line: "#302C30",
  placeholder1: "#702D3B",
  placeholder2: "#3B2028",
  placeholder3: "#343036",
  placeholder4: "#49313A",
};

const PLACEHOLDERS = [
  COLOURS.placeholder1,
  COLOURS.placeholder2,
  COLOURS.placeholder3,
  COLOURS.placeholder4,
];

function getBookId(item) {
  return (
    item.book_id ||
    item.bookRecord?.id ||
    item.book?.id ||
    item.id
  );
}

function getTitle(item) {
  return (
    item.title ||
    item.bookRecord?.title ||
    item.book?.title ||
    "Untitled"
  );
}

function getAuthor(item) {
  return (
    item.author ||
    item.bookRecord?.author ||
    item.book?.author ||
    ""
  );
}

function getCover(item) {
  return (
    item.editionRecord?.cover_url ||
    item.cover_url ||
    item.bookRecord?.cover_url ||
    item.book?.cover_url ||
    null
  );
}

function getSpine(item) {
  return (
    item.editionRecord?.spine_image_url ||
    item.spine_image_url ||
    item.bookRecord?.spine_image_url ||
    item.book?.spine_image_url ||
    null
  );
}

function Book({ item, index, onBookPress }) {
  const spine = getSpine(item);
  const cover = getCover(item);

  const widths = [34, 40, 36, 42, 38, 35];
  const heights = [145, 155, 160, 150, 170, 148];

  const width = widths[index % widths.length];
  const height = heights[index % heights.length];

  const faceOut =
    cover &&
    (index % 11 === 3 || index % 13 === 5);

  return (
    <Pressable
      onPress={() => onBookPress?.(item)}
      style={{
        marginHorizontal: 2,
        alignItems: "flex-end",
      }}
    >
      {faceOut ? (
        <Image
          source={{ uri: cover }}
          resizeMode="cover"
          style={{
            width: 68,
            height: 108,
            borderRadius: 3,
          }}
        />
      ) : spine ? (
        <Image
          source={{ uri: spine }}
          resizeMode="stretch"
          style={{
            width,
            height,
            borderRadius: 2,
          }}
        />
      ) : cover ? (
        <Image
          source={{ uri: cover }}
          resizeMode="cover"
          style={{
            width,
            height,
            borderRadius: 2,
          }}
        />
      ) : (
        <View
          style={{
            width,
            height,
            backgroundColor:
              PLACEHOLDERS[
                index % PLACEHOLDERS.length
              ],
            borderLeftWidth: 2,
            borderLeftColor:
              "rgba(255,255,255,0.08)",
            paddingHorizontal: 3,
            justifyContent: "center",
          }}
        >
          <Text
            numberOfLines={4}
            style={{
              color: COLOURS.text,
              fontSize: 7,
              textAlign: "center",
            }}
          >
            {getTitle(item)}
          </Text>
        </View>
      )}
    </Pressable>
  );
}

function Shelf({ books, shelfIndex, onBookPress }) {
  return (
    <View style={styles.shelfContainer}>
      <View style={styles.bookRow}>
        {books.map((book, index) => (
          <Book
            key={`${getBookId(book)}-${index}`}
            item={book}
            index={shelfIndex * 12 + index}
            onBookPress={onBookPress}
          />
        ))}
      </View>

      <View style={styles.shelfTop} />
      <View style={styles.shelfFace} />
    </View>
  );
}

export default function Bookshelf({
  items = [],
  onBookPress,
  emptyTitle = "Your library is waiting",
  emptyMessage = "Add books to start building your collection.",
}) {
  if (!items.length) {
    return (
      <View style={styles.empty}>
        <Text style={styles.emptyTitle}>
          {emptyTitle}
        </Text>

        <Text style={styles.emptyText}>
          {emptyMessage}
        </Text>
      </View>
    );
  }

  const shelves = [];

  for (let i = 0; i < items.length; i += 12) {
    shelves.push(items.slice(i, i + 12));
  }

  return (
    <View style={styles.container}>
      {shelves.map((shelf, index) => (
        <Shelf
          key={`shelf-${index}`}
          books={shelf}
          shelfIndex={index}
          onBookPress={onBookPress}
        />
      ))}
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    paddingBottom: 120,
  },

  shelfContainer: {
    marginBottom: 12,
  },

  bookRow: {
    minHeight: 180,
    flexDirection: "row",
    flexWrap: "wrap",
    alignItems: "flex-end",
    justifyContent: "center",
    paddingBottom: 6,
  },

  shelfTop: {
    height: 4,
    backgroundColor: COLOURS.shelfEdge,
  },

  shelfFace: {
    height: 10,
    backgroundColor: COLOURS.shelf,
    borderBottomWidth: 2,
    borderBottomColor: "#09090A",
  },

  empty: {
    backgroundColor: "#1C1B1E",
    borderWidth: 1,
    borderColor: COLOURS.line,
    borderRadius: 12,
    padding: 24,
    alignItems: "center",
    marginVertical: 20,
  },

  emptyTitle: {
    color: COLOURS.text,
    fontSize: 18,
    marginBottom: 8,
  },

  emptyText: {
    color: COLOURS.muted,
    textAlign: "center",
    lineHeight: 20,
  },
});
